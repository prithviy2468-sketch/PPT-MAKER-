    function getCandidateKeys(poolType) {
      // Pool routing with automatic cross-pool fallback if a dedicated key isn't entered yet
      let list = [];
      if (poolType === 'research') {
        list = [apiVault.researchPrimary, apiVault.researchBackup, apiVault.mainPrimary, apiVault.mainBackup];
      } else if (poolType === 'image') {
        list = [apiVault.imagePrimary, apiVault.imageBackup, apiVault.mainPrimary, apiVault.mainBackup];
      } else {
        list = [apiVault.mainPrimary, apiVault.mainBackup, apiVault.researchPrimary];
      }
      const unique = [...new Set(list.map(k => (k || '').trim()).filter(Boolean))];
      // Always include "" fallback for managed runtime environments where key is injected automatically
      if (!unique.length) unique.push('');
      return unique;
    }

    async function fetchWithPoolFailover(poolType, modelName, payloadBuilder, label = 'API') {
      const keys = getCandidateKeys(poolType);
      const pill = document.getElementById('active-api-pill');
      let lastErr = null;

      for (let kIdx = 0; kIdx < keys.length; kIdx++) {
        const key = keys[kIdx];
        const keyLabel = kIdx === 0 ? `${label} Primary` : `${label} Backup #${kIdx}`;
        if (pill) pill.textContent = `Active: ${keyLabel}`;

        const url = `https://generativelanguage.googleapis.com/v1beta/models/${modelName}:generateContent?key=${encodeURIComponent(key)}`;
        let delay = 900;

        for (let attempt = 0; attempt <= 2; attempt++) {
          try {
            const payload = payloadBuilder();
            const res = await fetch(url, {
              method: 'POST',
              headers: { 'Content-Type': 'application/json' },
              body: JSON.stringify(payload)
            });
            if (!res.ok) {
              const errBody = await res.text();
              throw new Error(`HTTP ${res.status}: ${errBody.slice(0, 120)}`);
            }
            const json = await res.json();
            recordApiUsage(JSON.stringify(payload).length, JSON.stringify(json).length);
            return json;
          } catch (err) {
            lastErr = err;
            if (attempt < 2) {
              await new Promise(r => setTimeout(r, delay));
              delay *= 2;
            }
          }
        }
        // If primary failed after retries, notify failover to Backup Key
        if (kIdx < keys.length - 1) {
          showToast(`⚠️ ${keyLabel} व्यस्त/त्रुटि — Backup API पर स्विच हो रहा है...`, 'info');
        }
      }
      throw lastErr || new Error('All API keys failed');
    }

    async function callGeminiJSON(promptText, schema, poolType = 'main', inlineParts = []) {
      const model = apiVault.mainModel || 'gemini-3-flash-preview';
      const parts = [{ text: promptText }, ...inlineParts];
      const result = await fetchWithPoolFailover(poolType, model, () => ({
        contents: [{ parts }],
        generationConfig: { responseMimeType: 'application/json', responseSchema: schema }
      }), poolType === 'research' ? 'Research' : 'Main');

      const txt = result?.candidates?.[0]?.content?.parts?.[0]?.text;
      if (!txt) throw new Error('Empty AI JSON response');
      return JSON.parse(txt);
    }

    async function callDeepResearch(topic, classLevel, langRule) {
      const prompt = `Conduct deep academic research on: "${topic}" for "${classLevel}".
Language: ${langRule}
Extract:
1. Core definitions, physical/chemical/biological mechanisms, and historical context.
2. Exact formulas, units, constants, and step-by-step mathematical derivations if applicable.
3. Board exam (UP Board / CBSE / NEET / JEE) frequently asked points and common misconceptions.
Provide structured, comprehensive research notes.`;

      const result = await fetchWithPoolFailover('research', apiVault.mainModel || 'gemini-3-flash-preview', () => {
        const body = { contents: [{ parts: [{ text: prompt }] }] };
        if (apiVault.useGoogleGrounding) body.tools = [{ google_search: {} }];
        return body;
      }, 'Research');

      return result?.candidates?.[0]?.content?.parts?.map(p => p.text).filter(Boolean).join('\n') || '';
    }

    async function callGeminiImageGen(imagePrompt, aspectRatio = '16:9') {
      const result = await fetchWithPoolFailover('image', 'gemini-3.1-flash-image', () => ({
        contents: [{
          role: 'user',
          parts: [{ text: `${imagePrompt}. Crisp, vibrant, high-resolution educational textbook diagram / scientific infographic, clean layout, sharp labels, no watermarks.` }]
        }],
        generationConfig: {
          responseModalities: ['IMAGE'],
          imageConfig: { aspectRatio }
        }
      }), 'Image');

      const part = result?.candidates?.[0]?.content?.parts?.find(p => p.inlineData);
      if (!part?.inlineData?.data) throw new Error('No image data returned');
      return `data:${part.inlineData.mimeType || 'image/png'};base64,${part.inlineData.data}`;
    }

    function openIDB() {
      return new Promise((resolve, reject) => {
        const req = indexedDB.open(IDB_NAME, 1);
        req.onupgradeneeded = (e) => {
          if (!e.target.result.objectStoreNames.contains(IDB_STORE)) e.target.result.createObjectStore(IDB_STORE);
        };
        req.onsuccess = () => resolve(req.result);
        req.onerror = () => reject(req.error);
      });
    }

    async function saveStateToIDB() {
      try {
        const db = await openIDB();
        const tx = db.transaction(IDB_STORE, 'readwrite');
        tx.objectStore(IDB_STORE).put(state, IDB_KEY);
        tx.oncomplete = () => {

        };
      } catch (e) {}
    }

    async function loadStateFromIDB() {
      try {
        const db = await openIDB();
        return await new Promise((resolve) => {
          const tx = db.transaction(IDB_STORE, 'readonly');
          const req = tx.objectStore(IDB_STORE).get(IDB_KEY);
          req.onsuccess = () => resolve(req.result || null);
          req.onerror = () => resolve(null);
        });
      } catch (e) { return null; }
    }

    function normalizeSlide(raw, idx = 0) {
      return {
        id: raw.id || `s-${Date.now()}-${idx}-${Math.random().toString(36).slice(2, 6)}`,
        layout: raw.layout || 'split-visual',
        badge: raw.badge || '',
        title: raw.title || '',
        subtitle: raw.subtitle || '',
        intro: raw.intro || '',
        bullets: Array.isArray(raw.bullets) ? raw.bullets : [],
        formula: raw.formula || '',
        derivation: raw.derivation && Array.isArray(raw.derivation.steps)
          ? raw.derivation
          : { given: 'दिया गया संबंध एवं परिभाषा:', steps: [{ title: 'चरण 1: मूल समीकरण', math: 'F = m × a' }, { title: 'चरण 2: मान रखने पर', math: 'W = F · d cos θ' }], result: 'सिद्ध परिणाम: W = F d cos θ' },
        flowchart: raw.flowchart && Array.isArray(raw.flowchart.nodes)
          ? raw.flowchart
          : { direction: 'horizontal', root: raw.title || 'मुख्य अवधारणा', nodes: [{ title: 'प्रकार / चरण 1', desc: 'संक्षिप्त विवरण' }, { title: 'प्रकार / चरण 2', desc: 'संक्षिप्त विवरण' }, { title: 'प्रकार / चरण 3', desc: 'संक्षिप्त विवरण' }] },
        table: raw.table || { headers: ['विषय', 'विवरण'], rows: [['बिंदु 1', 'मान 1']] },
        mcq: raw.mcq || { question: '', options: ['', '', '', ''], answerIndex: 0, explanation: '' },
        calloutTitle: raw.calloutTitle || '',
        calloutType: raw.calloutType || 'exam',
        calloutText: raw.calloutText || '',
        clipart: SVG_CLIPARTS[raw.clipart] ? raw.clipart : 'atom',
        pureImage: Boolean(raw.pureImage),
        imageFit: raw.imageFit === 'cover' ? 'cover' : 'contain',
        imageZoom: Number(raw.imageZoom) || 100,
        imagePanX: raw.imagePanX !== undefined ? Number(raw.imagePanX) : 50,
        imagePanY: raw.imagePanY !== undefined ? Number(raw.imagePanY) : 50,
        pins: Array.isArray(raw.pins) ? raw.pins : [],
        imagePrompt: raw.imagePrompt || '',
        imageCaption: raw.imageCaption || '',
        imageData: raw.imageData || null
      };
    }

    /***************************************************************************

    function buildSlideHTML(slide, index, total, interactive = true, showMcqAns = true) {
      const theme = getActiveTheme();
      const s = state.fontScale / 100;
      const ce = interactive ? 'contenteditable="true"' : '';

      // Global School / Coaching Logo & Watermark Lock Overlay
      const renderBrandingOverlay = () => {
        if (!state.branding?.locked) return '';
        let html = '';
        if (state.branding.logoData) {
          const posClass = state.branding.logoPos === 'top-left' ? 'top-3 left-4'
                         : state.branding.logoPos === 'bottom-left' ? 'bottom-11 left-4'

              <h2 class="font-extrabold leading-tight" style="font-size:${34 * s}px; color:${theme.accent};" ${ce} data-field="title">${escapeHtml(slide.title)}</h2>
              ${slide.subtitle ? `<p class="font-bold mt-1" style="font-size:${18 * s}px; color:${theme.titleColor};" ${ce} data-field="subtitle">${escapeHtml(slide.subtitle)}</p>` : ''}
            </div>`;
        }
        const titleBlock = theme.titleStyle === 'vertical-bar'
          ? `<div class="flex items-center gap-4">
               <div class="w-2 h-11 rounded-full shrink-0" style="background:${theme.accent};"></div>
               <div>
                 <h2 class="font-extrabold leading-tight" style="font-size:${30 * s}px; color:${theme.titleColor};" ${ce} data-field="title">${escapeHtml(slide.title)}</h2>
                 ${slide.subtitle ? `<p class="font-medium mt-0.5" style="font-size:${15.5 * s}px; color:${theme.subtitleColor};" ${ce} data-field="subtitle">${escapeHtml(slide.subtitle)}</p>` : ''}

             </div>`

             </div>`;

            <div class="min-w-0 flex-1">${titleBlock}</div>
            ${slide.badge ? `<div class="px-4 py-1.5 rounded-full text-xs font-bold shrink-0 shadow-sm" style="background:${theme.badgeBg}; color:${theme.badgeText}; border:1px solid ${theme.border};" ${ce} data-field="badge">${escapeHtml(slide.badge)}</div>` : ''}
          </div>`;
      };

      const renderFooter = () => `
        <div class="px-11 py-3 flex items-center justify-between text-xs shrink-0" style="border-top:1px solid ${theme.border}; color:${theme.mutedColor};">
          <span class="font-medium truncate max-w-[900px]">${escapeHtml(state.footerText)}</span>
          <span class="font-bold">${escapeHtml(state.pageLabel)} ${index + 1} / ${total}</span>
        </div>`;

      const renderVisualFrame = (heightPx = 410) => {
        const fit = slide.imageFit === 'cover' ? 'object-cover' : 'object-contain';
        const zoom = (slide.imageZoom || 100) / 100;
        const panX = slide.imagePanX ?? 50;
        const panY = slide.imagePanY ?? 50;
        const imgStyle = `object-position:${panX}% ${panY}%; transform:scale(${zoom}); transform-origin:${panX}% ${panY}%;`;

        const inner = slide.imageData
          ? `<img src="${slide.imageData}" alt="${escapeHtml(slide.imageCaption || slide.title)}" class="w-full h-full ${fit} rounded-xl transition-transform" style="${imgStyle}" />`
          : `<div class="w-full h-full flex flex-col items-center justify-center text-center p-6 gap-3 rounded-xl" style="background:${theme.accentSoft}; border:2px dashed ${theme.accent}55;">

              </div>
            </div>`).join('')}
        </div>`;

      // ==================== 1. FULL 16:9 IMAGE SLIDE ====================
      if (slide.layout === 'full-image') {
        const fitClass = slide.imageFit === 'cover' ? 'object-cover' : 'object-contain';
        const pure = Boolean(slide.pureImage);
        const zoom = (slide.imageZoom || 100) / 100;
        const panX = slide.imagePanX ?? 50;
        const panY = slide.imagePanY ?? 50;
        const imgStyle = `object-position:${panX}% ${panY}%; transform:scale(${zoom}); transform-origin:${panX}% ${panY}%;`;

        const pinsHTML = (slide.pins || []).map((pin, pi) => `
          <div class="callout-pin" data-pin-idx="${pi}" style="left:${pin.x}%; top:${pin.y}%;">

            ${pin.label ? `<div class="callout-pin-label">${escapeHtml(pin.label)}</div>` : ''}
          </div>`).join('');

        const mainVisual = slide.imageData
          ? `<img src="${slide.imageData}" alt="${escapeHtml(slide.imageCaption || slide.title)}" class="w-full h-full ${fitClass} block" style="${imgStyle}" />`

        return `
          <div class="w-full h-full relative overflow-hidden" style="background:${pure ? '#000000' : theme.bg};">

              </div>
              <div class="absolute bottom-4 left-6 right-6 z-20 flex items-center justify-between gap-4 px-5 py-2 rounded-2xl shadow-lg" style="background:rgba(15, 23, 42, 0.78); backdrop-filter:blur(8px); border:1px solid rgba(255,255,255,0.16);">

              <div class="space-y-4">
                ${nodes.filter((_, i) => i % 2 === 1).map((nd, idx) => {
                  const realIdx = idx * 2 + 1;
                  return `<div class="p-4 rounded-2xl shadow-sm" style="background:${theme.surface}; border:2px solid ${theme.quadColors[realIdx % 4]};">

    function fitStageToViewport(viewportId, outerId, innerId) {
      const vp = document.getElementById(viewportId);
      const outer = document.getElementById(outerId);
      const inner = document.getElementById(innerId);
      if (!vp || !outer || !inner) return;
      const st = getComputedStyle(vp);
      const w = vp.clientWidth - parseFloat(st.paddingLeft) - parseFloat(st.paddingRight);
      const h = vp.clientHeight - parseFloat(st.paddingTop) - parseFloat(st.paddingBottom);
      if (w <= 0 || h <= 0) return;
      const scale = Math.min(w / 1280, h / 720, 1);
      outer.style.width = `${Math.floor(1280 * scale)}px`;
      outer.style.height = `${Math.floor(720 * scale)}px`;
      inner.style.transform = `scale(${scale})`;
    }

    function attachPinDragging(stageEl) {
      stageEl.querySelectorAll('.callout-pin').forEach(pinEl => {
        const startDrag = (ev) => {
          ev.preventDefault();
          ev.stopPropagation();
          const pinIdx = parseInt(pinEl.dataset.pinIdx, 10);
          const outer = document.getElementById('stage-scaler-outer');
          const rect = outer.getBoundingClientRect();

          const moveDrag = (mEv) => {
            const clientX = mEv.touches ? mEv.touches[0].clientX : mEv.clientX;
            const clientY = mEv.touches ? mEv.touches[0].clientY : mEv.clientY;
            const xPct = Math.min(96, Math.max(4, ((clientX - rect.left) / rect.width) * 100));
            const yPct = Math.min(94, Math.max(6, ((clientY - rect.top) / rect.height) * 100));
            pinEl.style.left = `${xPct.toFixed(1)}%`;
            pinEl.style.top = `${yPct.toFixed(1)}%`;
            const curPin = state.slides[state.activeIndex].pins?.[pinIdx];
            if (curPin) {
              curPin.x = Number(xPct.toFixed(1));
              curPin.y = Number(yPct.toFixed(1));
            }
          };

          const endDrag = () => {
            window.removeEventListener('mousemove', moveDrag);
            window.removeEventListener('mouseup', endDrag);
            window.removeEventListener('touchmove', moveDrag);
            window.removeEventListener('touchend', endDrag);
            scheduleAutoSave();
          };

          window.addEventListener('mousemove', moveDrag);
          window.addEventListener('mouseup', endDrag);
          window.addEventListener('touchmove', moveDrag, { passive: false });
          window.addEventListener('touchend', endDrag);
        };
        pinEl.addEventListener('mousedown', startDrag);
        pinEl.addEventListener('touchstart', startDrag, { passive: false });
      });
    }

    function renderAll() {
      if (state.activeIndex >= state.slides.length) state.activeIndex = Math.max(0, state.slides.length - 1);
      renderSlideList();
      renderActiveSlide();

      syncInspectorForms();
      updateAppendPreview();
      fitStageToViewport('stage-viewport', 'stage-scaler-outer', 'slide-stage-1280');
      scheduleAutoSave();

    }

    function renderSlideList() {
      const list = document.getElementById('slide-list');
      const q = (document.getElementById('input-slide-search')?.value || '').trim().toLowerCase();
      const theme = getActiveTheme();
      const cur = state.slides[state.activeIndex];

      document.getElementById('slide-count-badge').textContent = state.slides.length;
      document.getElementById('active-slide-pill').textContent = `${state.pageLabel} ${state.activeIndex + 1} / ${state.slides.length}`;
      document.getElementById('active-layout-pill').textContent = cur?.layout || 'split-visual';
      document.getElementById('active-theme-indicator').textContent = theme.name;
      document.getElementById('input-slide-jump').max = state.slides.length;

      const items = state.slides.map((sl, idx) => ({ sl, idx })).filter(({ sl, idx }) => {
        if (!q) return true;
        return (sl.title || '').toLowerCase().includes(q) || (sl.layout || '').toLowerCase().includes(q) || String(idx + 1) === q;
      });

      list.innerHTML = items.map(({ sl, idx }) => {
        const active = idx === state.activeIndex;
        const badgeCol = sl.layout === 'full-image' ? 'bg-amber-500/20 text-amber-300'
                       : sl.layout === 'derivation' ? 'bg-emerald-500/20 text-emerald-300'
                       : sl.layout === 'flowchart' ? 'bg-purple-500/20 text-purple-300'
                       : 'bg-slate-800 text-slate-400';
        return `
          <button onclick="selectSlide(${idx})" class="w-44 lg:w-full shrink-0 text-left rounded-xl p-2 transition border flex items-center gap-2.5 ${active ? 'border-orange-500 bg-orange-500/15 shadow-sm' : 'border-slate-800 bg-slate-900/90 hover:border-slate-600'}">
            <div class="w-7 h-7 rounded-lg flex items-center justify-center text-xs font-extrabold shrink-0" style="background:${active ? '#f97316' : '#1e293b'}; color:${active ? '#fff' : '#94a3b8'};">${idx + 1}</div>
            <div class="min-w-0 flex-1">
              <div class="text-[11px] font-bold text-slate-100 truncate">${escapeHtml(sl.title || 'Untitled')}</div>
              <div class="flex items-center gap-1.5 mt-0.5">
                <span class="text-[9px] uppercase px-1.5 rounded font-bold ${badgeCol}">${sl.layout}</span>
                ${sl.imageData ? '<span class="text-[9px] text-emerald-400">🖼️</span>' : ''}
              </div>

    function renderActiveSlide() {
      const stage = document.getElementById('slide-stage-1280');
      const slide = state.slides[state.activeIndex];
      if (!slide) return;
      stage.innerHTML = buildSlideHTML(slide, state.activeIndex, state.slides.length, true, mcqAnswerVisible);
      attachPinDragging(stage);

      stage.querySelectorAll('[contenteditable="true"]').forEach(el => {
        el.addEventListener('focus', () => { lastFocusedField = el; });
        el.addEventListener('input', () => {
          const cur = state.slides[state.activeIndex];
          const f = el.dataset.field;
          const idx = parseInt(el.dataset.idx, 10);
          const val = el.innerText.trim();

          if (f === 'title') cur.title = val;
          else if (f === 'subtitle') cur.subtitle = val;
          else if (f === 'badge') cur.badge = val;

    function insertSymbolAtCursor(sym) {
      const el = lastFocusedField;
      if (!el) { showToast('पहले किसी टेक्स्ट बॉक्स पर क्लिक करें।', 'info'); return; }
      if (el.tagName === 'INPUT' || el.tagName === 'TEXTAREA') {
        const st = el.selectionStart ?? el.value.length;
        const en = el.selectionEnd ?? el.value.length;
        el.value = el.value.slice(0, st) + sym + el.value.slice(en);
        el.focus();
        el.setSelectionRange(st + sym.length, st + sym.length);
        el.dispatchEvent(new Event('input', { bubbles: true }));
      } else if (el.isContentEditable) {
        el.focus();
        document.execCommand('insertText', false, sym);
        el.dispatchEvent(new Event('input', { bubbles: true }));
      }
    }

    async function renderPdfPageToDataURL(pdfDoc, pageNum) {
      const page = await pdfDoc.getPage(pageNum);
      const viewport = page.getViewport({ scale: 1.5 });
      const canvas = document.createElement('canvas');
      canvas.width = viewport.width;
      canvas.height = viewport.height;
      await page.render({ canvasContext: canvas.getContext('2d'), viewport }).promise;
      return canvas.toDataURL('image/jpeg', 0.88);
    }

          const dataUrl = await renderPdfPageToDataURL(loadedPdfDoc, p);
          imageParts.push({ inlineData: { mimeType: 'image/jpeg', data: dataUrl.split(',')[1] } });

        const mime = scannedBookImageDataURL.split(';')[0].split(':')[1] || 'image/jpeg';

      if (!topic) { showToast('कृपया टॉपिक या अध्याय का नाम लिखें।', 'error'); return; }

      closeAIModal();
      const langRule = LANGUAGE_INSTRUCTIONS[langKey] || LANGUAGE_INSTRUCTIONS.bilingual;
      const styleRule = styleMode === 'pairs'
        ? 'IMPORTANT PATTERN: Alternate every concept as a 2-slide pair -> Odd content slides MUST use layout "full-image" (with a detailed 16:9 widescreen educational diagram English imagePrompt), followed immediately by Even content slides using layout "text-focus" or "derivation".'
        : 'Use a rich academic mix across layouts: "cover", "full-image", "text-focus", "derivation", "flowchart", "split-visual", "table-visual", "formula", "steps", "quiz", and "mindmap".';

      let researchContext = '';
      if (useResearch) {
        openProgress('🔬 Deep Research API सक्रिय...', `"${topic}" पर गहन रिसर्च व सूत्र संकलन...`, 8);
        try {
          researchContext = await callDeepResearch(topic, classLevel, langRule);
        } catch (e) {}
      }

      if (aiModalMode === 'batch') {
        const count = parseInt(document.getElementById('ai-batch-count').value, 10) || 12;
        await generateBatchSlides({ topic, classLevel, langRule, styleRule, researchContext, count, mode, autoImgs });
      } else {
        const targetTotal = parseInt(document.getElementById('ai-mega-count').value, 10) || 50;
        await run50PageMegaBuilder({ topic, classLevel, langRule, styleRule, researchContext, targetTotal, mode, autoImgs });
      }
    }

    async function generateBatchSlides({ topic, classLevel, langRule, styleRule, researchContext, count, mode, autoImgs }) {
      openProgress('AI बैच स्लाइड्स बना रहा है...', `Main API ${count} स्लाइड्स तैयार कर रहा है...`, 18);
      const batchSchema = {
        type: "OBJECT",
        properties: { footer: { type: "STRING" }, slides: { type: "ARRAY", items: SLIDE_SCHEMA } },
        required: ["footer", "slides"]
      };

      const prompt = `Generate exactly ${count} consistent academic slides on: "${topic}" (${classLevel}).
${researchContext ? `Research Notes:\n${researchContext.slice(0, 4000)}\n` : ''}
Language Requirement: ${langRule}
Layout Pattern: ${styleRule}
- ${mode === 'new' ? 'Slide 1 MUST be "cover".' : 'Appending to existing deck: do not repeat a cover slide.'}
- Include detailed English "imagePrompt" for every visual/full-image slide.`;

      try {
        const res = await callGeminiJSON(prompt, batchSchema, 'main');
        if (cancelRequested) { closeProgress(); return; }
        const built = (res.slides || []).map((s, i) => normalizeSlide(s, i));
        if (!built.length) throw new Error('Empty slides');

        const startIdx = mode === 'append' ? state.slides.length : 0;
        if (mode === 'append') state.slides.push(...built);
        else { state.slides = built; if (res.footer) state.footerText = res.footer; }
        state.activeIndex = startIdx;
        renderAll();

        if (autoImgs) {
          const imgIndices = built
            .map((s, i) => (!['text-focus', 'derivation', 'flowchart', 'quiz', 'steps', 'mindmap'].includes(s.layout) ? startIdx + i : -1))
            .filter(i => i >= 0);
          await generateImagesForIndices(imgIndices);
        }

        closeProgress();
        showToast(`${built.length} स्लाइड्स तैयार! कुल: ${state.slides.length}`, 'success');
      } catch (e) {
        closeProgress();
        showToast('AI बैच बनाने में त्रुटि आई। API Vault जाँचें।', 'error');
      }
    }

    async function run50PageMegaBuilder({ topic, classLevel, langRule, styleRule, researchContext, targetTotal, mode, autoImgs }) {
      openProgress('🚀 50-Page Mega Builder', `चरण 1: ${targetTotal} स्लाइड्स की अध्याय रूपरेखा बन रही है...`, 12);
      const outlineSchema = {
        type: "OBJECT",
        properties: {
          footer: { type: "STRING" },
          outline: {
            type: "ARRAY",
            items: {
              type: "OBJECT",
              properties: {
                slideNo: { type: "INTEGER" },
                title: { type: "STRING" },
                layout: { type: "STRING", enum: ["cover", "full-image", "text-focus", "derivation", "flowchart", "split-visual", "simple-box", "table-visual", "formula", "steps", "quiz", "mindmap"] },
                focus: { type: "STRING" }
              },
              required: ["slideNo", "title", "layout", "focus"]
            }
          }
        },
        required: ["footer", "outline"]
      };

      const outlinePrompt = `Plan a comprehensive ${targetTotal}-slide academic deck on: "${topic}" (${classLevel}).
${researchContext ? `Research Notes:\n${researchContext.slice(0, 3500)}\n` : ''}
Language: ${langRule}
Layout Pattern: ${styleRule}
Return exactly ${targetTotal} items in "outline".`;

      try {
        const plan = await callGeminiJSON(outlinePrompt, outlineSchema, 'main');
        if (cancelRequested) { closeProgress(); return; }
        const outline = plan.outline || [];
        if (!outline.length) throw new Error('Outline failed');

        if (mode === 'new') {
          state.slides = [];
          if (plan.footer) state.footerText = plan.footer;
          state.activeIndex = 0;
        }

        const startIdxAll = state.slides.length;
        const batchSize = 8;
        const totalBatches = Math.ceil(outline.length / batchSize);
        const batchSchema = { type: "OBJECT", properties: { slides: { type: "ARRAY", items: SLIDE_SCHEMA } }, required: ["slides"] };

        let created = 0;
        for (let b = 0; b < totalBatches; b++) {
          if (cancelRequested) break;
          const chunk = outline.slice(b * batchSize, (b + 1) * batchSize);
          updateProgress(`बैच ${b + 1} / ${totalBatches}: स्लाइड ${b * batchSize + 1}–${b * batchSize + chunk.length} बन रही हैं...`, 16 + Math.round((b / totalBatches) * 80));

          const chunkPrompt = `Generate detailed academic slides for this batch of ${chunk.length} slides from "${topic}" (${classLevel}).
Language: ${langRule}
Follow this exact outline: ${JSON.stringify(chunk)}`;

          const res = await callGeminiJSON(chunkPrompt, batchSchema, 'main');
          const built = (res.slides || []).map((s, idx) => normalizeSlide(s, idx));
          state.slides.push(...built);
          created += built.length;
          if (state.slides.length === built.length) state.activeIndex = 0;
          renderAll();
        }

        if (autoImgs && !cancelRequested && created > 0) {
          const imgIndices = [];
          for (let i = startIdxAll; i < startIdxAll + created; i++) {
            const l = state.slides[i]?.layout;
            if (l && !['text-focus', 'derivation', 'flowchart', 'quiz', 'steps', 'mindmap'].includes(l)) imgIndices.push(i);
          }
          await generateImagesForIndices(imgIndices);
        }

        closeProgress();
        showToast(`मेगा बिल्डर पूर्ण! ${created} स्लाइड्स बनीं (कुल: ${state.slides.length})`, 'success');
      } catch (e) {
        closeProgress();
        showToast('मेगा बिल्डर रुका — अब तक बनी स्लाइड्स सुरक्षित हैं!', 'error');
      }
    }

    async function generateCurrentSlideImage() {
      const cur = state.slides[state.activeIndex];
      const promptText = document.getElementById('input-img-prompt').value.trim() || cur.imagePrompt || cur.title;
      cur.imagePrompt = promptText;

      const loader = document.getElementById('slide-img-loading');
      const btn = document.getElementById('btn-gen-slide-img');
      loader.classList.remove('hidden'); btn.disabled = true;

      try {
        const aspect = cur.layout === 'full-image' ? '16:9' : '4:3';
        cur.imageData = await callGeminiImageGen(promptText, aspect);
        renderAll();
        showToast(`${aspect} AI इमेज तैयार है!`, 'success');
      } catch (e) {
        cur.imageData = buildFallbackDiagramDataURL(cur, getActiveTheme());
        renderAll();
        showToast('स्मार्ट 16:9 वेक्टर डायग्राम लगाया गया।', 'info');
      } finally {
        loader.classList.add('hidden'); btn.disabled = false;
      }
    }

    async function generateImagesForIndices(indices) {
      if (!indices.length) { showToast('कोई खाली इमेज स्लाइड नहीं मिली।', 'info'); return; }
      openProgress('बैच AI इमेज बन रही हैं...', `0 / ${indices.length} पूर्ण`, 5);
      let done = 0;
      for (const idx of indices) {
        if (cancelRequested) break;
        const sl = state.slides[idx];
        if (!sl) continue;
        done++;
        const aspect = sl.layout === 'full-image' ? '16:9' : '4:3';
        updateProgress(`इमेज ${done} / ${indices.length} (पृष्ठ ${idx + 1} • ${aspect})...`, Math.round((done / indices.length) * 100));
        try {
          sl.imageData = await callGeminiImageGen(sl.imagePrompt || sl.title, aspect);
        } catch (e) {
          sl.imageData = buildFallbackDiagramDataURL(sl, getActiveTheme());
        }
        renderAll();
      }
      closeProgress();
      showToast(`${done} स्लाइड्स की AI इमेज तैयार हो गई!`, 'success');
    }

    /***************************************************************************
     * 11. EDITABLE .PPTX EXPORT (All 12 Layouts + Branding + Derivation + Flowchart)
     ***************************************************************************/
    function exportSlidesToPPTX(slidesArray, filenamePrefix = 'AI_PPT_Studio_v5') {
      if (typeof PptxGenJS === 'undefined') { showToast('PptxGenJS लोड हो रहा है...', 'error'); return; }
      try {
        const pptx = new PptxGenJS();
        pptx.layout = 'LAYOUT_16x9';
        pptx.title = slidesArray[0]?.title || filenamePrefix;

        const theme = getActiveTheme();
        const hex = c => c.replace('#', '');
        const total = slidesArray.length;

        slidesArray.forEach((s, idx) => {
          const slide = pptx.addSlide();
          slide.background = { color: s.layout === 'full-image' && s.pureImage ? '000000' : hex(theme.bg) };

          // Global Branding Overlay
          if (state.branding?.locked) {
            if (state.branding.logoData && state.branding.logoData.startsWith('data:image/')) {
              const lx = state.branding.logoPos === 'top-left' ? 0.35 : (state.branding.logoPos === 'bottom-left' ? 0.35 : 8.55);
              const ly = state.branding.logoPos === 'bottom-left' ? 4.65 : 0.15;
              try { slide.addImage({ data: state.branding.logoData, x: lx, y: ly, w: 1.1, h: 0.45, sizing: { type: 'contain', w: 1.1, h: 0.45 } }); } catch (e) {}
            }
            if (state.branding.watermarkText) {
              slide.addText(state.branding.watermarkText, { x: 7.2, y: 4.88, w: 2.4, h: 0.25, fontSize: 8.5, align: 'right', color: hex(theme.mutedColor), fontFace: 'Nirmala UI' });
            }
          }

          // 1. FULL 16:9 IMAGE
          if (s.layout === 'full-image') {
            if (s.imageData && s.imageData.startsWith('data:image/')) {
              try {
                slide.addImage({
                  data: s.imageData, x: 0, y: 0, w: 10, h: 5.625,
                  sizing: { type: s.imageFit === 'cover' ? 'cover' : 'contain', w: 10, h: 5.625 }
                });
              } catch (e) {}
            }
            (s.pins || []).forEach((pin, pi) => {
              const px = Math.max(0.05, (pin.x / 100) * 10 - 0.16);
              const py = Math.max(0.05, (pin.y / 100) * 5.625 - 0.16);
              slide.addShape(pptx.ShapeType.ellipse, { x: px, y: py, w: 0.32, h: 0.32, fill: { color: 'EA580C' }, line: { color: 'FFFFFF', width: 1.5 } });
              slide.addText(String(pin.num || pi + 1), { x: px, y: py, w: 0.32, h: 0.32, fontSize: 10, bold: true, color: 'FFFFFF', align: 'center', valign: 'middle', fontFace: 'Nirmala UI' });
              if (pin.label) slide.addText(pin.label, { x: Math.min(7.5, px + 0.36), y: py, w: 2.3, h: 0.32, fontSize: 9, bold: true, color: 'FFFFFF', fill: { color: '0F172A' }, fontFace: 'Nirmala UI' });
            });
            if (!s.pureImage) {
              slide.addShape(pptx.ShapeType.roundRect, { x: 0.3, y: 0.18, w: 9.4, h: 0.48, fill: { color: '0F172A', transparency: 22 } });
              slide.addText(s.title || '', { x: 0.45, y: 0.22, w: 7.5, h: 0.4, fontSize: 15, bold: true, color: 'FFFFFF', fontFace: 'Nirmala UI' });
              slide.addShape(pptx.ShapeType.roundRect, { x: 0.3, y: 5.05, w: 9.4, h: 0.4, fill: { color: '0F172A', transparency: 22 } });
              slide.addText(s.imageCaption || state.footerText || '', { x: 0.45, y: 5.09, w: 7.7, h: 0.32, fontSize: 10, color: 'F1F5F9', fontFace: 'Nirmala UI' });
              slide.addText(`${state.pageLabel} ${idx + 1} / ${total}`, { x: 8.2, y: 5.09, w: 1.35, h: 0.32, fontSize: 9.5, bold: true, align: 'right', color: 'FDE047', fontFace: 'Nirmala UI' });
            }
            return;
          }

          // Standard Footer
          slide.addShape(pptx.ShapeType.line, { x: 0.45, y: 5.18, w: 9.1, h: 0, line: { color: hex(theme.border), width: 0.75 } });
          slide.addText(state.footerText || '', { x: 0.45, y: 5.23, w: 7.4, h: 0.28, fontSize: 9.5, color: hex(theme.mutedColor), fontFace: 'Nirmala UI' });
          slide.addText(`${state.pageLabel} ${idx + 1} / ${total}`, { x: 8.1, y: 5.23, w: 1.45, h: 0.28, fontSize: 9.5, bold: true, align: 'right', color: hex(theme.mutedColor), fontFace: 'Nirmala UI' });

          const addVisual = (x, y, w, h, capY) => {
            slide.addShape(pptx.ShapeType.roundRect, { x, y, w, h, fill: { color: hex(theme.surface) }, line: { color: hex(theme.border), width: 1 } });
            if (s.imageData && s.imageData.startsWith('data:image/')) {
              try { slide.addImage({ data: s.imageData, x: x + 0.08, y: y + 0.08, w: w - 0.16, h: h - 0.16, sizing: { type: s.imageFit === 'cover' ? 'cover' : 'contain', w: w - 0.16, h: h - 0.16 } }); } catch (e) {}
            }
            if (s.imageCaption) slide.addText(s.imageCaption, { x, y: capY, w, h: 0.28, fontSize: 9.5, italic: true, align: 'center', color: hex(theme.mutedColor), fontFace: 'Nirmala UI' });
          };

          const addCallout = (x, y, w, h) => {
            if (!s.calloutText) return;
            const cs = s.calloutType === 'simple' ? theme.calloutSimple : theme.calloutExam;
            slide.addShape(pptx.ShapeType.rect, { x, y, w, h, fill: { color: hex(cs.bg) }, line: { color: hex(cs.border), width: 1 } });
            slide.addText([
              { text: (s.calloutTitle || 'मुख्य बिंदु') + '\n', options: { bold: true, color: hex(cs.title), fontSize: 10.5 } },
              { text: s.calloutText, options: { color: hex(cs.text), fontSize: 9.5 } }
            ], { x: x + 0.12, y: y + 0.04, w: w - 0.2, h: h - 0.08, valign: 'top', fontFace: 'Nirmala UI' });
          };

          if (s.layout === 'cover') {
            slide.addShape(pptx.ShapeType.rect, { x: 0, y: 0, w: 10, h: 0.12, fill: { color: hex(theme.accent) } });
            if (s.badge) slide.addText(s.badge, { x: 0.55, y: 0.7, w: 4.5, h: 0.35, fontSize: 11, bold: true, color: hex(theme.badgeText), fill: { color: hex(theme.badgeBg) }, fontFace: 'Nirmala UI' });
            slide.addText(s.title || '', { x: 0.55, y: 1.2, w: 5.3, h: 1.25, fontSize: 27, bold: true, color: hex(theme.titleColor), fontFace: 'Nirmala UI' });
            slide.addText(s.subtitle || '', { x: 0.55, y: 2.65, w: 5.3, h: 0.65, fontSize: 14.5, bold: true, color: hex(theme.accent), fontFace: 'Nirmala UI' });
            if (s.intro) slide.addText(s.intro, { x: 0.55, y: 3.4, w: 5.3, h: 1.1, fontSize: 12, color: hex(theme.mutedColor), fontFace: 'Nirmala UI' });
            addVisual(6.1, 0.85, 3.45, 3.6, 4.52);
            return;
          }

          // Header
          slide.addText(s.title || '', { x: 0.45, y: 0.24, w: 6.9, h: 0.48, fontSize: 20, bold: true, color: hex(theme.titleColor), fontFace: 'Nirmala UI' });
          if (s.badge) slide.addText(s.badge, { x: 7.45, y: 0.28, w: 2.1, h: 0.32, fontSize: 9.5, bold: true, align: 'center', color: hex(theme.badgeText), fill: { color: hex(theme.badgeBg) }, fontFace: 'Nirmala UI' });

          // 2. DERIVATION
          if (s.layout === 'derivation') {
            const d = s.derivation || { given: '', steps: [], result: '' };
            let dy = 0.95;
            if (d.given) {
              slide.addText(`दिया है: ${d.given}`, { x: 0.45, y: dy, w: 9.1, h: 0.42, fontSize: 11, bold: true, color: hex(theme.titleColor), fill: { color: hex(theme.surface) }, line: { color: hex(theme.border), width: 1 }, fontFace: 'Nirmala UI' });
              dy += 0.52;
            }
            (d.steps || []).slice(0, 5).forEach((st, si) => {
              slide.addShape(pptx.ShapeType.roundRect, { x: 0.45, y: dy, w: 9.1, h: 0.44, fill: { color: hex(theme.surface) }, line: { color: hex(theme.quadColors[si % 4]), width: 1 } });
              slide.addText(`${si + 1}. ${st.title}`, { x: 0.58, y: dy + 0.04, w: 5.3, h: 0.36, fontSize: 10.5, bold: true, color: hex(theme.titleColor), fontFace: 'Nirmala UI' });
              slide.addText(st.math, { x: 5.95, y: dy + 0.04, w: 3.45, h: 0.36, fontSize: 11, bold: true, align: 'right', color: hex(theme.accent), fontFace: 'Consolas' });
              dy += 0.5;
            });
            if (d.result) {
              slide.addText(d.result, { x: 0.45, y: Math.min(dy + 0.05, 3.68), w: 9.1, h: 0.44, fontSize: 12.5, bold: true, align: 'center', color: hex(theme.tableHeaderText), fill: { color: hex(theme.tableHeaderBg) }, fontFace: 'Nirmala UI' });
            }
            addCallout(0.45, 4.22, 9.1, 0.84);
            return;
          }

          // 3. FLOWCHART
          if (s.layout === 'flowchart') {
            const fc = s.flowchart || { root: s.title, nodes: [] };
            slide.addText(fc.root || s.title, { x: 2.2, y: 0.98, w: 5.6, h: 0.45, fontSize: 13.5, bold: true, align: 'center', color: hex(theme.tableHeaderText), fill: { color: hex(theme.tableHeaderBg) }, fontFace: 'Nirmala UI' });
            const nodes = (fc.nodes || []).slice(0, 4);
            const colW = nodes.length ? (9.1 - (nodes.length - 1) * 0.2) / nodes.length : 4;
            nodes.forEach((nd, ni) => {
              const nx = 0.45 + ni * (colW + 0.2);
              slide.addShape(pptx.ShapeType.roundRect, { x: nx, y: 1.68, w: colW, h: 2.3, fill: { color: hex(theme.surface) }, line: { color: hex(theme.quadColors[ni % 4]), width: 1.5 } });

    function initSmartBoardInteractions() {
      const canvas = document.getElementById('smartboard-canvas');
      const overlay = document.getElementById('present-overlay');
      const laser = document.getElementById('laser-dot');
      const spot = document.getElementById('spotlight-overlay');
      const outer = document.getElementById('present-scaler-outer');

      const trackPointer = (e) => {
        const cx = e.touches ? e.touches[0].clientX : e.clientX;
        const cy = e.touches ? e.touches[0].clientY : e.clientY;
        if (sbTool === 'laser') {
          laser.style.left = `${cx}px`;
          laser.style.top = `${cy}px`;
        } else if (sbTool === 'spotlight') {
          const r = outer.getBoundingClientRect();
          const rx = cx - r.left;
          const ry = cy - r.top;
          spot.style.background = `radial-gradient(circle 130px at ${rx}px ${ry}px, transparent 0%, rgba(0, 0, 0, 0.82) 100%)`;
        }
      };
      overlay.addEventListener('mousemove', trackPointer);
      overlay.addEventListener('touchmove', trackPointer, { passive: true });

      const getPos = (e) => {
        const rect = canvas.getBoundingClientRect();
        const cx = e.touches ? e.touches[0].clientX : e.clientX;
        const cy = e.touches ? e.touches[0].clientY : e.clientY;
        return { x: ((cx - rect.left) / rect.width) * canvas.width, y: ((cy - rect.top) / rect.height) * canvas.height };
      };

      const start = (e) => {
        if (!['pen', 'highlighter', 'eraser'].includes(sbTool) || !sbCtx) return;
        e.preventDefault(); isDrawing = true;
        const p = getPos(e);
        sbCtx.beginPath(); sbCtx.moveTo(p.x, p.y);
      };
      const move = (e) => {
        if (!isDrawing || !sbCtx) return;
        e.preventDefault();
        const p = getPos(e);
        if (sbTool === 'eraser') { sbCtx.globalCompositeOperation = 'destination-out'; sbCtx.lineWidth = 26; sbCtx.globalAlpha = 1; }
        else if (sbTool === 'highlighter') { sbCtx.globalCompositeOperation = 'source-over'; sbCtx.strokeStyle = sbColor; sbCtx.lineWidth = 20; sbCtx.globalAlpha = 0.28; }
        else { sbCtx.globalCompositeOperation = 'source-over'; sbCtx.strokeStyle = sbColor; sbCtx.lineWidth = 3.5; sbCtx.globalAlpha = 1; }
        sbCtx.lineTo(p.x, p.y); sbCtx.stroke();
      };
      const end = () => { isDrawing = false; if (sbCtx) sbCtx.closePath(); };

      canvas.addEventListener('mousedown', start); canvas.addEventListener('mousemove', move); window.addEventListener('mouseup', end);
      canvas.addEventListener('touchstart', start, { passive: false }); canvas.addEventListener('touchmove', move, { passive: false }); window.addEventListener('touchend', end);
    }

    function openProjectJSON(file, mergeMode = false) {
      const r = new FileReader();
      r.onload = (e) => {
        try {
          const loaded = JSON.parse(e.target.result);
          if (!Array.isArray(loaded.slides) || !loaded.slides.length) throw new Error('Invalid');
          const normalized = loaded.slides.map((s, i) => normalizeSlide(s, i));
          if (mergeMode) {
            state.slides.push(...normalized);
            renderAll();
            showToast(`Merge सफल! ${normalized.length} नई स्लाइड्स जुड़ीं (कुल: ${state.slides.length})`, 'success');
          } else {
            state.themeId = getAllThemes()[loaded.themeId] ? loaded.themeId : 'up-board';
            if (loaded.customThemes) state.customThemes = loaded.customThemes;
            if (loaded.branding) state.branding = loaded.branding;
            state.footerText = loaded.footerText || state.footerText;
            state.pageLabel = loaded.pageLabel || 'पृष्ठ';
            state.fontScale = loaded.fontScale || 100;
            state.slides = normalized;
            state.activeIndex = 0;
            renderAll();
            showToast(`प्रोजेक्ट लोड हो गया (${state.slides.length} स्लाइड्स)!`, 'success');
          }
        } catch (err) {
          showToast('अमान्य प्रोजेक्ट फाइल।', 'error');
        }
      };
      r.readAsText(file);
    }
